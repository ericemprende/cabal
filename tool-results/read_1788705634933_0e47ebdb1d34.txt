   468→ * Gestor de wallets conectadas + verificación de tokens antiguos como dev.
   469→ * - Conectar: wallet inyectada (Phantom/MetaMask) o dirección pegada a mano.
   470→ * - Verificar posesión: firma criptográfica (ed25519 / personal_sign).
   471→ * - Track record: reclama un CA y el backend trae métricas reales on-chain
   472→ *   (MC, ATH, liquidez, volumen, concentración) vía DexScreener/GeckoTerminal.
   473→ */
   474→function WalletManager({ me }: { me: MeDTO }) {
   475→  const [network, setNetwork] = useState('solana')
   476→  const [address, setAddress] = useState('')
   477→  const [walletKind, setWalletKind] = useState<'phantom' | 'evm' | null>(null)
   478→
   479→  useEffect(() => {
   480→    // async: las wallets inyectadas solo existen en el cliente
   481→    const t = setTimeout(() => setWalletKind(injectedWalletFor(network)), 0)
   482→    return () => clearTimeout(t)
   483→  }, [network])
   484→
   485→  const addWallet = useAddWallet()
   486→  const verifySig = useVerifyWalletSignature()
   487→  const removeWallet = useRemoveWallet()
   488→  const verifyToken = useVerifyDevToken()
   489→  const removeClaim = useRemoveDevToken()
   490→
   491→  const connectInjected = async () => {
   492→    try {
   493→      if (walletKind === 'phantom') {
   494→        const provider = window.phantom?.solana ?? window.solana
   495→        if (!provider?.connect) throw new Error('No se detectó Phantom')
   496→        const res = await provider.connect()
   497→        setAddress(res.publicKey.toString())
   498→      } else if (walletKind === 'evm') {
   499→        const accounts = (await window.ethereum?.request({ method: 'eth_requestAccounts' })) as
   500→          | string[]
   501→          | undefined
   502→        if (accounts?.[0]) setAddress(accounts[0])
   503→      } else {
   504→        toast.error('No hay wallet del navegador para esta red; pega la dirección a mano')
   505→      }
   506→    } catch (e) {
   507→      toast.error((e as Error).message || 'Conexión rechazada')
   508→    }
   509→  }
   510→
   511→  const add = () => {
   512→    const a = address.trim()
   513→    if (!a) {
   514→      toast.error('Pega la dirección de tu wallet')
   515→      return
   516→    }
   517→    addWallet.mutate(
   518→      {
   519→        network,
   520→        address: a,
   521→        label: walletKind === 'phantom' ? 'Phantom' : walletKind === 'evm' ? 'MetaMask' : 'Manual',
   522→      },
   523→      { onSuccess: () => setAddress('') }
   524→    )
   525→  }
   526→
   527→  const unsigned = me.wallets.filter((w) => !w.signature)
   528→
   529→  return (
   530→    <div className="space-y-3">
   531→      {/* Wallets conectadas */}
   532→      {me.wallets.length > 0 && (
   533→        <div className="space-y-1.5">
   534→          {me.wallets.map((w) => (
   535→            <WalletRow
   536→              key={w.id}
   537→              wallet={w}
   538→              onRemove={() => removeWallet.mutate(w.id)}
   539→              onVerify={() => verifySig.mutate({ id: w.id, network: w.network, address: w.address })}
   540→              verifying={verifySig.isPending}
   541→              busy={removeWallet.isPending}
   542→            />
   543→          ))}
   544→        </div>
   545→      )}
   546→
   547→      {/* Conectar nueva wallet */}
   548→      <div className="rounded-xl border border-white/10 bg-[#0a0b08] p-3">
   549→        <p className="flex items-center gap-1.5 text-[13px] font-bold">
   550→          <Wallet className="h-3.5 w-3.5 text-primary" aria-hidden /> Conectar wallet
   551→        </p>
   552→        <div className="mt-2 flex flex-wrap gap-1.5" role="radiogroup" aria-label="Red de la wallet">
   553→          {Object.entries(NETWORKS).map(([key, meta]) => (
   554→            <button
   555→              key={key}
   556→              type="button"
   557→              onClick={() => setNetwork(key)}
   558→              aria-pressed={network === key}
   559→              className={cn(
   560→                'flex items-center gap-1 rounded-full border px-2.5 py-1 text-[11px] font-semibold transition-all',
   561→                network === key
   562→                  ? 'border-[#8FA83F]/50 bg-[#8FA83F]/10 text-primary'
   563→                  : 'border-white/10 text-muted-foreground hover:border-white/25 hover:text-foreground'
   564→              )}
   565→            >
   566→              <NetworkIcon network={key} className="h-3 w-3" />
   567→              {meta.label}
   568→            </button>
   569→          ))}
   570→        </div>
   571→        <div className="mt-2.5 flex items-center gap-1.5">
   572→          <Input
   573→            value={address}
   574→            onChange={(e) => setAddress(e.target.value)}
   575→            placeholder={`Dirección ${NETWORKS[network as keyof typeof NETWORKS]?.label ?? network}…`}
   576→            inputMode="text"
   577→            spellCheck={false}
   578→            autoComplete="off"
   579→            aria-label={`Dirección de wallet en ${network}`}
   580→            className="h-9 min-w-0 flex-1 bg-[#121410] font-mono text-xs"
   581→          />
   582→          {walletKind && (
   583→            <Button
   584→              size="sm"
   585→              variant="ghost"
   586→              onClick={connectInjected}
   587→              className="h-9 shrink-0 rounded-lg border border-[#8FA83F]/35 bg-[#8FA83F]/10 px-2.5 text-xs font-bold text-primary hover:bg-[#8FA83F]/20 hover:text-primary"
   588→            >
   589→              {walletKind === 'phantom' ? 'Phantom' : 'MetaMask'}
   590→            </Button>
   591→          )}
   592→          <Button
   593→            size="sm"
   594→            onClick={add}
   595→            disabled={addWallet.isPending}
   596→            className="h-9 shrink-0 rounded-lg bg-primary px-3 text-xs font-bold text-primary-foreground hover:bg-[#8FA83F]"
   597→          >
   598→            {addWallet.isPending ? '…' : 'Conectar'}
   599→          </Button>
   600→        </div>
   601→        {unsigned.length > 0 && (
   602→          <p className="mt-2 text-[10px] leading-relaxed text-muted-foreground">
   603→            Firma la wallet para probar que eres el dueño (+10 pts, una vez). Si no tienes la
   604→            extensión a mano, la dirección queda guardada igualmente.
   605→          </p>
   606→        )}
   607→      </div>
   608→
   609→      {/* Verificar tokens antiguos como dev */}
   610→      <DevClaimForm
   611→        me={me}
   612→        onVerify={(data) => verifyToken.mutate(data)}
   613→        verifying={verifyToken.isPending}
   614→      />
   615→
   616→      {/* Track record */}
   617→      {me.devClaims.length > 0 ? (
   618→        <div className="space-y-2">
   619→          {me.devClaims.map((claim) => (
   620→            <ClaimCard
   621→              key={claim.id}
   622→              claim={claim}
   623→              onRetry={() =>
   624→                verifyToken.mutate({
   625→                  network: claim.network,
   626→                  contract: claim.contract,
   627→                  walletAddress: claim.walletAddress,
   628→                })
   629→              }
   630→              onDelete={() => removeClaim.mutate(claim.id)}
   631→              busy={verifyToken.isPending || removeClaim.isPending}
   632→            />
   633→          ))}
   634→        </div>
   635→      ) : (
   636→        <p className="rounded-xl border border-dashed border-white/10 px-3 py-3 text-[11px] leading-relaxed text-muted-foreground">
   637→          Lanzaste tokens antes de Cabal? Pega el CA y verifica on-chain que fuiste el dev:
   638→          tu perfil mostrará el ATH, la liquidez y la concentración real de esos tokens.
   639→        </p>
   640→      )}
   641→    </div>
   642→  )
   643→}
