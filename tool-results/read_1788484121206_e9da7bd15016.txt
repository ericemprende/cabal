    74→function toInputDateTime(iso: string): string {
    75→  const d = new Date(iso)
    76→  const pad = (n: number) => String(n).padStart(2, '0')
    77→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`
    78→}

    81→function ChipToggle({
    82→  label,
    83→  checked,
    84→  onChange,
    85→  okIcon = true,
    86→}: {
    87→  label: string
    88→  checked: boolean
    89→  onChange: (v: boolean) => void
    90→  okIcon?: boolean
    91→}) {
    92→  return (
    93→    <button
    94→      type="button"
    95→      onClick={() => onChange(!checked)}
    96→      aria-pressed={checked}
    97→      className={cn(
    98→        'inline-flex items-center gap-1.5 rounded-lg border px-2.5 py-1.5 text-[11px] font-bold transition-all',
    99→        checked
   100→          ? 'border-[#8FA83F]/45 bg-[#8FA83F]/12 text-primary'
   101→          : 'border-white/10 bg-[#121410] text-muted-foreground hover:border-white/25'
   102→      )}
   103→    >
   104→      {okIcon &&
   105→        (checked ? (
   106→          <CheckCircle2 className="h-3 w-3" aria-hidden />
   107→        ) : (
   108→          <XCircle className="h-3 w-3" aria-hidden />
   109→        ))}
   110→      {label}
   111→    </button>
   112→  )
   113→}

   309→// ---------------- Usuarios: puntos + edición de perfil ----------------
   310→function AdminUserRow({
   311→  user,
   312→  enabled,
   313→  onAdjust,
   314→}: {
   315→  user: AdminUserRowDTO
   316→  enabled: boolean
   317→  onAdjust: (amount: number, note?: string) => void
   318→}) {
   319→  const updateUser = useAdminUpdateUser(enabled)
   320→  const [amount, setAmount] = useState('50')
   321→  const [note, setNote] = useState('')
   322→  const [editing, setEditing] = useState(false)
   323→  const [form, setForm] = useState({
   324→    name: user.name,
   325→    handle: user.handle,
   326→    xHandle: user.xHandle ?? '',
   327→    tgHandle: user.tgHandle ?? '',
   328→    googleEmail: user.googleEmail ?? '',
   329→    walletVerified: user.walletVerified,
   330→    xVerified: user.xVerified,
   331→    googleVerified: user.googleVerified,
   332→    isDev: user.isDev,
   333→    isAdmin: user.isAdmin ?? false,
   334→  })
   335→  const amt = parseInt(amount, 10) || 0
   336→  const set = (k: string, v: string | boolean) => setForm((f) => ({ ...f, [k]: v }))
   337→
   338→  const doAdjust = (sign: 1 | -1) => {
   339→    if (amt <= 0) {
   340→      toast.error('Ingresa un monto mayor a 0')
   341→      return
   342→    }
   343→    onAdjust(sign * amt, note.trim() || undefined)
   344→    setNote('')
   345→    toast.success(`${sign > 0 ? '+' : '-'}${amt} puntos para @${user.handle}`)
   346→  }
   347→
   348→  const save = () => {
   349→    updateUser.mutate(
   350→      {
   351→        id: user.id,
   352→        name: form.name,
   353→        handle: form.handle,
   354→        xHandle: form.xHandle,
   355→        tgHandle: form.tgHandle,
   356→        googleEmail: form.googleEmail,
   357→        walletVerified: form.walletVerified,
   358→        xVerified: form.xVerified,
   359→        googleVerified: form.googleVerified,
   360→        isDev: form.isDev,
   361→        isAdmin: form.isAdmin,
   362→      },
   363→      { onSuccess: () => setEditing(false) }
   364→    )
   365→  }
   366→
   367→  return (
   368→    <div className="rounded-xl border border-white/10 bg-[#0a0b08] p-3">
   369→      <div className="flex flex-wrap items-center gap-2.5">
   370→        <UserAvatar name={user.name} handle={user.handle} size="md" verified={user.walletVerified} />
   371→        <div className="min-w-0 flex-1">
   372→          <p className="truncate text-sm font-bold">
   373→            {user.name}
   374→            {user.isAdmin && <span className="ml-1.5 rounded bg-[#8FA83F]/12 px-1 py-px text-[9px] font-black text-primary">ADMIN</span>}
   375→            {user.xVerified && <span className="ml-1 rounded bg-white/8 px-1 py-px text-[9px] font-black text-zinc-300">X</span>}
   376→            {user.googleVerified && <span className="ml-1 rounded bg-white/8 px-1 py-px text-[9px] font-black text-zinc-300">G</span>}
   377→          </p>
   378→          <p className="flex items-center gap-1 truncate text-xs text-muted-foreground">
   379→            @{user.handle} · {user.postsCount} posts · {user.launchesCount} launches · {user.likesReceived}
   380→            <Heart className="h-3 w-3" aria-hidden />
   381→          </p>
   382→        </div>
   383→        <PointsPill points={user.points} />
   384→        <Button
   385→          size="sm"
   386→          variant="ghost"
   387→          onClick={() => setEditing((v) => !v)}
   388→          className="h-8 gap-1.5 rounded-lg border border-white/10 px-2.5 text-xs font-semibold text-muted-foreground hover:text-foreground"
   389→        >
   390→          <Pencil className="h-3 w-3" /> {editing ? 'Cerrar' : 'Editar'}
   391→        </Button>
   392→        <div className="flex items-center gap-1.5">
   393→          <Input
   394→            value={note}
   395→            onChange={(e) => setNote(e.target.value)}
   396→            placeholder="Motivo (opcional)"
   397→            className="h-8 w-32 border-white/10 bg-[#121410] text-xs"
   398→            aria-label="Motivo del ajuste"
   399→          />
   400→          <Input
   401→            type="number"
   402→            value={amount}
   403→            onChange={(e) => setAmount(e.target.value)}
   404→            className="h-8 w-16 border-white/10 bg-[#121410] text-center font-mono text-xs font-bold"
   405→            aria-label="Monto"
   406→          />
   407→          <Button size="icon" onClick={() => doAdjust(1)} className="h-8 w-8 rounded-lg bg-primary text-primary-foreground hover:bg-[#8FA83F]" aria-label="Añadir puntos">
   408→            <Plus className="h-4 w-4" strokeWidth={3} />
   409→          </Button>
   410→          <Button size="icon" variant="secondary" onClick={() => doAdjust(-1)} className="h-8 w-8 rounded-lg text-[#ff8080] hover:bg-destructive/15" aria-label="Quitar puntos">
   411→            <Minus className="h-4 w-4" strokeWidth={3} />
   412→          </Button>
   413→        </div>
   414→      </div>
   415→
   416→      {editing && (
   417→        <div className="mt-3 space-y-3 rounded-lg border border-white/8 bg-[#121410] p-3">
   418→          <div className="grid gap-2.5 sm:grid-cols-2">
   419→            <div className="space-y-1">
   420→              <Label className="text-[10px] uppercase tracking-wide text-muted-foreground">Nombre</Label>
   421→              <Input value={form.name} onChange={(e) => set('name', e.target.value)} className="h-8 bg-[#0a0b08] text-[13px]" />
   422→            </div>
   423→            <div className="space-y-1">
   424→              <Label className="text-[10px] uppercase tracking-wide text-muted-foreground">Handle</Label>
   425→              <Input value={form.handle} onChange={(e) => set('handle', e.target.value)} className="h-8 bg-[#0a0b08] font-mono text-[13px]" />
   426→            </div>
   427→            <div className="space-y-1">
   428→              <Label className="text-[10px] uppercase tracking-wide text-muted-foreground">X (Twitter)</Label>
   429→              <Input value={form.xHandle} onChange={(e) => set('xHandle', e.target.value)} placeholder="@usuario" className="h-8 bg-[#0a0b08] text-[13px]" />
   430→            </div>
   431→            <div className="space-y-1">
   432→              <Label className="text-[10px] uppercase tracking-wide text-muted-foreground">Telegram</Label>
   433→              <Input value={form.tgHandle} onChange={(e) => set('tgHandle', e.target.value)} placeholder="@usuario" className="h-8 bg-[#0a0b08] text-[13px]" />
   434→            </div>
   435→            <div className="space-y-1 sm:col-span-2">
   436→              <Label className="text-[10px] uppercase tracking-wide text-muted-foreground">Google (email)</Label>
   437→              <Input value={form.googleEmail} onChange={(e) => set('googleEmail', e.target.value)} placeholder="usuario@gmail.com" className="h-8 bg-[#0a0b08] text-[13px]" />
   438→            </div>
   439→          </div>
   440→          <div className="flex flex-wrap gap-1.5">
   441→            <ChipToggle label="Wallet verificada" checked={form.walletVerified} onChange={(v) => set('walletVerified', v)} />
   442→            <ChipToggle label="X verificada" checked={form.xVerified} onChange={(v) => set('xVerified', v)} />
   443→            <ChipToggle label="Google verificada" checked={form.googleVerified} onChange={(v) => set('googleVerified', v)} />
   444→            <ChipToggle label="Dev" checked={form.isDev} onChange={(v) => set('isDev', v)} okIcon={false} />
   445→            <ChipToggle label="Admin" checked={form.isAdmin} onChange={(v) => set('isAdmin', v)} okIcon={false} />
   446→          </div>
   447→          <div className="flex justify-end">
   448→            <Button
   449→              size="sm"
   450→              onClick={save}
