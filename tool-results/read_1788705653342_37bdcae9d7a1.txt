   101→function toInputDateTime(iso: string): string {
   102→  const d = new Date(iso)
   103→  const pad = (n: number) => String(n).padStart(2, '0')
   104→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`
   105→}

   108→function ChipToggle({
   109→  label,
   110→  checked,
   111→  onChange,
   112→  okIcon = true,
   113→}: {
   114→  label: string
   115→  checked: boolean
   116→  onChange: (v: boolean) => void
   117→  okIcon?: boolean
   118→}) {
   119→  return (
   120→    <button
   121→      type="button"
   122→      onClick={() => onChange(!checked)}
   123→      aria-pressed={checked}
   124→      className={cn(
   125→        'inline-flex items-center gap-1.5 rounded-lg border px-2.5 py-1.5 text-[11px] font-bold transition-all',
   126→        checked
   127→          ? 'border-[#8FA83F]/45 bg-[#8FA83F]/12 text-primary'
   128→          : 'border-white/10 bg-[#121410] text-muted-foreground hover:border-white/25'
   129→      )}
   130→    >
   131→      {okIcon &&
   132→        (checked ? (
   133→          <CheckCircle2 className="h-3 w-3" aria-hidden />
   134→        ) : (
   135→          <XCircle className="h-3 w-3" aria-hidden />
   136→        ))}
   137→      {label}
   138→    </button>
   139→  )
   140→}

   362→// ---------------- Usuarios: puntos + edición de perfil ----------------
   363→function AdminUserRow({
   364→  user,
   365→  enabled,
   366→  onAdjust,
   367→}: {
   368→  user: AdminUserRowDTO
   369→  enabled: boolean
   370→  onAdjust: (amount: number, note?: string) => void
   371→}) {
   372→  const updateUser = useAdminUpdateUser(enabled)
   373→  const [amount, setAmount] = useState('50')
   374→  const [note, setNote] = useState('')
   375→  const [editing, setEditing] = useState(false)
   376→  const [form, setForm] = useState({
   377→    name: user.name,
   378→    handle: user.handle,
   379→    xHandle: user.xHandle ?? '',
   380→    tgHandle: user.tgHandle ?? '',
   381→    googleEmail: user.googleEmail ?? '',
   382→    walletVerified: user.walletVerified,
   383→    xVerified: user.xVerified,
   384→    googleVerified: user.googleVerified,
   385→    isDev: user.isDev,
   386→    isAdmin: user.isAdmin ?? false,
   387→  })
   388→  const amt = parseInt(amount, 10) || 0
   389→  const set = (k: string, v: string | boolean) => setForm((f) => ({ ...f, [k]: v }))
   390→
   391→  const doAdjust = (sign: 1 | -1) => {
   392→    if (amt <= 0) {
   393→      toast.error('Ingresa un monto mayor a 0')
   394→      return
   395→    }
   396→    onAdjust(sign * amt, note.trim() || undefined)
   397→    setNote('')
   398→    toast.success(`${sign > 0 ? '+' : '-'}${amt} puntos para @${user.handle}`)
   399→  }
   400→
   401→  const save = () => {
   402→    updateUser.mutate(
   403→      {
   404→        id: user.id,
   405→        name: form.name,
   406→        handle: form.handle,
   407→        xHandle: form.xHandle,
   408→        tgHandle: form.tgHandle,
   409→        googleEmail: form.googleEmail,
   410→        walletVerified: form.walletVerified,
   411→        xVerified: form.xVerified,
   412→        googleVerified: form.googleVerified,
   413→        isDev: form.isDev,
   414→        isAdmin: form.isAdmin,
   415→      },
   416→      { onSuccess: () => setEditing(false) }
   417→    )
   418→  }
   419→
   420→  return (
   421→    <div className="rounded-xl border border-white/10 bg-[#0a0b08] p-3">
   422→      <div className="flex flex-wrap items-center gap-2.5">
   423→        <UserAvatar name={user.name} handle={user.handle} src={user.avatar} size="md" verified={user.walletVerified} />
   424→        <div className="min-w-0 flex-1">
   425→          <p className="truncate text-sm font-bold">
   426→            {user.name}
   427→            {user.isAdmin && <span className="ml-1.5 rounded bg-[#8FA83F]/12 px-1 py-px text-[9px] font-black text-primary">ADMIN</span>}
   428→            {user.xVerified && <span className="ml-1 rounded bg-white/8 px-1 py-px text-[9px] font-black text-zinc-300">X</span>}
   429→            {user.googleVerified && <span className="ml-1 rounded bg-white/8 px-1 py-px text-[9px] font-black text-zinc-300">G</span>}
   430→          </p>
   431→          <p className="flex items-center gap-1 truncate text-xs text-muted-foreground">
   432→            @{user.handle} · {user.postsCount} posts · {user.launchesCount} launches · {user.likesReceived}
   433→            <Heart className="h-3 w-3" aria-hidden />
   434→          </p>
   435→        </div>
   436→        <PointsPill points={user.points} />
   437→        <Button
   438→          size="sm"
   439→          variant="ghost"
   440→          onClick={() => setEditing((v) => !v)}
   441→          className="h-8 gap-1.5 rounded-lg border border-white/10 px-2.5 text-xs font-semibold text-muted-foreground hover:text-foreground"
   442→        >
   443→          <Pencil className="h-3 w-3" /> {editing ? 'Cerrar' : 'Editar'}
   444→        </Button>
   445→        <div className="flex items-center gap-1.5">
   446→          <Input
   447→            value={note}
   448→            onChange={(e) => setNote(e.target.value)}
   449→            placeholder="Motivo (opcional)"
   450→            className="h-8 w-32 border-white/10 bg-[#121410] text-xs"
   451→            aria-label="Motivo del ajuste"
   452→          />
   453→          <Input
   454→            type="number"
   455→            value={amount}
   456→            onChange={(e) => setAmount(e.target.value)}
   457→            className="h-8 w-16 border-white/10 bg-[#121410] text-center font-mono text-xs font-bold"
   458→            aria-label="Monto"
   459→          />
   460→          <Button size="icon" onClick={() => doAdjust(1)} className="h-8 w-8 rounded-lg bg-primary text-primary-foreground hover:bg-[#8FA83F]" aria-label="Añadir puntos">
   461→            <Plus className="h-4 w-4" strokeWidth={3} />
   462→          </Button>
   463→          <Button size="icon" variant="secondary" onClick={() => doAdjust(-1)} className="h-8 w-8 rounded-lg text-[#ff8080] hover:bg-destructive/15" aria-label="Quitar puntos">
   464→            <Minus className="h-4 w-4" strokeWidth={3} />
   465→          </Button>
   466→        </div>
   467→      </div>
   468→
   469→      {editing && (
   470→        <div className="mt-3 space-y-3 rounded-lg border border-white/8 bg-[#121410] p-3">
   471→          <div className="grid gap-2.5 sm:grid-cols-2">
   472→            <div className="space-y-1">
   473→              <Label className="text-[10px] uppercase tracking-wide text-muted-foreground">Nombre</Label>
