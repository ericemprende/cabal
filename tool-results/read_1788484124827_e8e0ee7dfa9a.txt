   310→   310→function AdminUserRow({
   311→   311→  user,
   312→   312→  enabled,
   313→   313→  onAdjust,
   314→   314→}: {
   315→   315→  user: AdminUserRowDTO
   316→   316→  enabled: boolean
   317→   317→  onAdjust: (amount: number, note?: string) => void
   318→   318→}) {
   319→   319→  const updateUser = useAdminUpdateUser(enabled)
   320→   320→  const [amount, setAmount] = useState('50')
   321→   321→  const [note, setNote] = useState('')
   322→   322→  const [editing, setEditing] = useState(false)
   323→   323→  const [form, setForm] = useState({
   324→   324→    name: user.name,
   325→   325→    handle: user.handle,
   326→   326→    xHandle: user.xHandle ?? '',
   327→   327→    tgHandle: user.tgHandle ?? '',
   328→   328→    googleEmail: user.googleEmail ?? '',
   329→   329→    walletVerified: user.walletVerified,
   330→   330→    xVerified: user.xVerified,
   331→   331→    googleVerified: user.googleVerified,
   332→   332→    isDev: user.isDev,
   333→   333→    isAdmin: user.isAdmin ?? false,
   334→   334→  })
   335→   335→  const amt = parseInt(amount, 10) || 0
   336→   336→  const set = (k: string, v: string | boolean) => setForm((f) => ({ ...f, [k]: v }))
   337→   337→
   338→   338→  const doAdjust = (sign: 1 | -1) => {
   339→   339→    if (amt <= 0) {
   340→   340→      toast.error('Ingresa un monto mayor a 0')
   341→   341→      return
   342→   342→    }
   343→   343→    onAdjust(sign * amt, note.trim() || undefined)
   344→   344→    setNote('')
   345→   345→    toast.success(`${sign > 0 ? '+' : '-'}${amt} puntos para @${user.handle}`)
   346→   346→  }
   347→   347→
   348→   348→  const save = () => {
   349→   349→    updateUser.mutate(
   350→   350→      {
   351→   351→        id: user.id,
   352→   352→        name: form.name,
   353→   353→        handle: form.handle,
   354→   354→        xHandle: form.xHandle,
   355→   355→        tgHandle: form.tgHandle,
   356→   356→        googleEmail: form.googleEmail,
   357→   357→        walletVerified: form.walletVerified,
   358→   358→        xVerified: form.xVerified,
   359→   359→        googleVerified: form.googleVerified,
   360→   360→        isDev: form.isDev,
   361→   361→        isAdmin: form.isAdmin,
   362→   362→      },
   363→   363→      { onSuccess: () => setEditing(false) }
   364→   364→    )
   365→   365→  }
   366→   366→
   367→   367→  return (
   368→   368→    <div className="rounded-xl border border-white/10 bg-[#0a0b08] p-3">
   369→   369→      <div className="flex flex-wrap items-center gap-2.5">
   370→   370→        <UserAvatar name={user.name} handle={user.handle} size="md" verified={user.walletVerified} />
   371→   371→        <div className="min-w-0 flex-1">
   372→   372→          <p className="truncate text-sm font-bold">
   373→   373→            {user.name}
   374→   374→            {user.isAdmin && <span className="ml-1.5 rounded bg-[#8FA83F]/12 px-1 py-px text-[9px] font-black text-primary">ADMIN</span>}
   375→   375→            {user.xVerified && <span className="ml-1 rounded bg-white/8 px-1 py-px text-[9px] font-black text-zinc-300">X</span>}
   376→   376→            {user.googleVerified && <span className="ml-1 rounded bg-white/8 px-1 py-px text-[9px] font-black text-zinc-300">G</span>}
   377→   377→          </p>
   378→   378→          <p className="flex items-center gap-1 truncate text-xs text-muted-foreground">
   379→   379→            @{user.handle} · {user.postsCount} posts · {user.launchesCount} launches · {user.likesReceived}
   380→   380→            <Heart className="h-3 w-3" aria-hidden />
   381→   381→          </p>
   382→   382→        </div>
   383→   383→        <PointsPill points={user.points} />
   384→   384→        <Button
   385→   385→          size="sm"
   386→   386→          variant="ghost"
   387→   387→          onClick={() => setEditing((v) => !v)}
   388→   388→          className="h-8 gap-1.5 rounded-lg border border-white/10 px-2.5 text-xs font-semibold text-muted-foreground hover:text-foreground"
   389→   389→        >
   390→   390→          <Pencil className="h-3 w-3" /> {editing ? 'Cerrar' : 'Editar'}
   391→   391→        </Button>
   392→   392→        <div className="flex items-center gap-1.5">
   393→   393→          <Input
   394→   394→            value={note}
   395→   395→            onChange={(e) => setNote(e.target.value)}
   396→   396→            placeholder="Motivo (opcional)"
   397→   397→            className="h-8 w-32 border-white/10 bg-[#121410] text-xs"
   398→   398→            aria-label="Motivo del ajuste"
   399→   399→          />
   400→   400→          <Input
   401→   401→            type="number"
   402→   402→            value={amount}
   403→   403→            onChange={(e) => setAmount(e.target.value)}
